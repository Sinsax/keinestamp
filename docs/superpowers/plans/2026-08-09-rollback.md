# Rollback 回退/恢复 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用"文件真回退 + 最新状态暂存"的 rollback 机制替换旧的 revert/restore，并重做 GUI 入口（右键菜单 + 恢复按钮）。

**Architecture:** 回退状态存在项目自己的 `.git` 里：两个 git ref（`refs/keinestamp/rollback` = 回退前 HEAD、`refs/keinestamp/rollback-point` = 回退点）+ 可选 stash（message 前缀 `keinestamp-rollback`，存工作区差异）。回退 = stash + `git restore --source=<X>`；恢复 = stash pop（无 stash 时 restore 回 HEAD）；冲突 = GUI 文件二选一（`git show <stash>:<path>` 写回）。零全局状态（AppConfig 不再存 last_revert）。

**Tech Stack:** Rust 2024（anyhow/clap/serde/tauri v2）、Svelte 5 + TypeScript。

## Global Constraints

- git 安全：禁止 checkout/branch/merge/rebase/reset --hard/force push。新增例外：`git stash push/pop`、`git restore --source=<commit> --staged --worktree`、`git update-ref refs/keinestamp/*`、`git show <stash>:<path>` 写回文件（全部本地、可逆、不碰历史指针）
- stash message 前缀固定为 `keinestamp-rollback`
- 中文 UI/CLI 文案（已撤回 → 回退、恢复）
- 验证命令：`cargo test`、`cd frontend; npm run check`
- 日志格式 `eprintln!("[debug] ...")` 风格沿用现有代码

```rust
const ROLLBACK_REF: &str = "refs/keinestamp/rollback";
const ROLLBACK_POINT_REF: &str = "refs/keinestamp/rollback-point";
const ROLLBACK_STASH_PREFIX: &str = "keinestamp-rollback";

fn find_rollback_stash(project_root: &Path) -> Result<Option<String>> {
    let out = run_git_output(project_root, &["stash", "list", "--format=%gd %s"])?;
    Ok(out
        .lines()
        .find_map(|line| {
            let mut it = line.splitn(2, ' ');
            let refname = it.next()?;
            let msg = it.next().unwrap_or("");
            if msg.starts_with(ROLLBACK_STASH_PREFIX) {
                Some(refname.to_string())
            } else {
                None
            }
        }))
}

fn run_git_bytes(path: &Path, args: &[&str]) -> Result<Vec<u8>> {
    let output = crate::process_util::hide_console(Command::new("git"))
        .args(args)
        .current_dir(path)
        .output()
        .with_context(|| format!("failed to execute git {:?} at {:?}", args, path))?;
    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        anyhow::bail!("git {} failed: {}", args.join(" "), stderr.trim());
    }
    Ok(output.stdout)
}

fn clear_rollback_refs(project_root: &Path) -> Result<()> {
    let _ = run_git(project_root, &["update-ref", "-d", ROLLBACK_REF]);
    let _ = run_git(project_root, &["update-ref", "-d", ROLLBACK_POINT_REF]);
    Ok(())
}
```

```rust
pub fn rollback(project_root: &Path, entry_hash: &str) -> Result<()> {
    eprintln!("[debug] rollback: root={:?}, entry={}", project_root, entry_hash);
    let head = get_current_commit_hash(project_root)?;

    let dirty = run_git_output(project_root, &["status", "--porcelain"])?
        .lines()
        .any(|l| !l.trim().is_empty());
    if dirty {
        run_git(project_root, &["stash", "push", "-u", "-m", ROLLBACK_STASH_PREFIX])?;
    }

    run_git(
        project_root,
        &["restore", "--source", entry_hash, "--staged", "--worktree", "--", "."],
    )?;
    run_git(project_root, &["update-ref", ROLLBACK_REF, &head])?;
    run_git(project_root, &["update-ref", ROLLBACK_POINT_REF, entry_hash])?;
    eprintln!("[debug] rollback → ok");
    Ok(())
}

pub struct RollbackState {
    pub rollback_point: Option<String>,
    pub head_matches: bool,
}

pub fn get_rollback_state(project_root: &Path) -> Result<RollbackState> {
    let point = run_git_output(project_root, &["rev-parse", "--verify", ROLLBACK_POINT_REF])
        .ok()
        .map(|s| s.trim().to_string())
        .filter(|s| !s.is_empty());
    let Some(point) = point else {
        return Ok(RollbackState { rollback_point: None, head_matches: false });
    };
    let saved_head = run_git_output(project_root, &["rev-parse", "--verify", ROLLBACK_REF])?;
    let head = get_current_commit_hash(project_root)?;
    Ok(RollbackState {
        rollback_point: Some(point),
        head_matches: head == saved_head.trim(),
    })
}

pub fn restore_rollback(project_root: &Path) -> Result<()> {
    eprintln!("[debug] restore_rollback: root={:?}", project_root);
    let saved_head = run_git_output(project_root, &["rev-parse", "--verify", ROLLBACK_REF])?;
    match find_rollback_stash(project_root)? {
        Some(stash_ref) => run_git(project_root, &["stash", "pop", &stash_ref])?,
        None => run_git(
            project_root,
            &["restore", "--source", saved_head.trim(), "--staged", "--worktree", "--", "."],
        )?,
    }
    clear_rollback_refs(project_root)?;
    eprintln!("[debug] restore_rollback → ok");
    Ok(())
}

pub fn get_rollback_conflicts(project_root: &Path) -> Result<Vec<String>> {
    let out = run_git_output(project_root, &["diff", "--name-only", "--diff-filter=U"])?;
    Ok(out.lines().map(|s| s.trim().to_string()).filter(|s| !s.is_empty()).collect())
}

pub fn resolve_rollback_conflicts(project_root: &Path, choices: &[(String, bool)]) -> Result<()> {
    eprintln!("[debug] resolve_rollback_conflicts: root={:?}, {} files", project_root, choices.len());
    let stash_ref = find_rollback_stash(project_root)?
        .ok_or_else(|| anyhow::anyhow!("没有找到回退暂存（stash）"))?;
    for (path, use_latest) in choices {
        if *use_latest {
            let content = run_git_bytes(project_root, &["show", &format!("{}:{}", stash_ref, path)])?;
            let full = project_root.join(path);
            if let Some(parent) = full.parent() {
                std::fs::create_dir_all(parent)?;
            }
            std::fs::write(&full, &content)?;
        }
        run_git(project_root, &["add", path])?;
    }
    run_git(project_root, &["stash", "drop", &stash_ref])?;
    clear_rollback_refs(project_root)?;
    eprintln!("[debug] resolve_rollback_conflicts → ok");
    Ok(())
}

pub fn clear_rollback(project_root: &Path) -> Result<()> {
    if let Some(s) = find_rollback_stash(project_root)? {
        let _ = run_git(project_root, &["stash", "drop", &s]);
    }
    clear_rollback_refs(project_root)
}
```

```rust
#[tauri::command]
async fn rollback(path: String, entry_id: String) -> Result<(), String> {
    eprintln!("[debug] rollback({}, {})", path, &entry_id[..7.min(entry_id.len())]);
    let handle = tokio::task::spawn_blocking(move || {
        let p = PathBuf::from(&path);
        let meta_path = keinestamp::config::meta_path(&p);
        let meta = map_err(keinestamp::config::read_meta(&meta_path))?;
        map_err(keinestamp::git_project::rollback(&meta.project_root, &entry_id))?;
        Ok(())
    });
    handle.await.map_err(|e| e.to_string())?
}

#[tauri::command]
fn get_rollback_state(path: String) -> Result<serde_json::Value, String> {
    eprintln!("[debug] get_rollback_state({})", path);
    let p = PathBuf::from(&path);
    let meta_path = keinestamp::config::meta_path(&p);
    let meta = map_err(keinestamp::config::read_meta(&meta_path))?;
    let st = map_err(keinestamp::git_project::get_rollback_state(&meta.project_root))?;
    Ok(serde_json::json!({
        "rollback_point": st.rollback_point,
        "head_matches": st.head_matches,
    }))
}

#[tauri::command]
async fn restore_rollback(path: String) -> Result<(), String> {
    eprintln!("[debug] restore_rollback({})", path);
    let handle = tokio::task::spawn_blocking(move || {
        let p = PathBuf::from(&path);
        let meta_path = keinestamp::config::meta_path(&p);
        let meta = map_err(keinestamp::config::read_meta(&meta_path))?;
        match keinestamp::git_project::restore_rollback(&meta.project_root) {
            Ok(_) => Ok(()),
            Err(e) => {
                if let Ok(files) = keinestamp::git_project::get_rollback_conflicts(&meta.project_root) {
                    if !files.is_empty() {
                        return Err(format!("CONFLICT|{}", files.join("\n")));
                    }
                }
                Err(e.to_string())
            }
        }
    });
    handle.await.map_err(|e| e.to_string())?
}

#[tauri::command]
fn get_rollback_conflicts(path: String) -> Result<Vec<String>, String> {
    eprintln!("[debug] get_rollback_conflicts({})", path);
    let p = PathBuf::from(&path);
    let meta_path = keinestamp::config::meta_path(&p);
    let meta = map_err(keinestamp::config::read_meta(&meta_path))?;
    map_err(keinestamp::git_project::get_rollback_conflicts(&meta.project_root))
}

#[tauri::command]
async fn resolve_rollback_conflicts(path: String, choices: Vec<Vec<String>>) -> Result<(), String> {
    eprintln!("[debug] resolve_rollback_conflicts({}, {} files)", path, choices.len());
    let handle = tokio::task::spawn_blocking(move || {
        let p = PathBuf::from(&path);
        let meta_path = keinestamp::config::meta_path(&p);
        let meta = map_err(keinestamp::config::read_meta(&meta_path))?;
        let parsed: Vec<(String, bool)> = choices
            .into_iter()
            .map(|c| {
                let path = c.first().cloned().unwrap_or_default();
                let use_latest = c.get(1).map(|s| s == "latest").unwrap_or(false);
                (path, use_latest)
            })
            .collect();
        map_err(keinestamp::git_project::resolve_rollback_conflicts(&meta.project_root, &parsed))
    });
    handle.await.map_err(|e| e.to_string())?
}
```

```ts
  } catch { /* ignore */ }

async function handleRollback(entry: TimelineEntry) {
  if (!activePath || rollbackBusy) return
  rollbackBusy = true
  try {
    await api.rollback(activePath, entry.id)
    pushToast(`已回退到记录 ${entry.id.slice(0, 7)}（文件已还原）`, 'success')
    await loadProject(activePath)
    await refreshRollbackState()
  } catch (e: any) {
    pushToast(`回退失败: ${e}`, 'error')
  }
  rollbackBusy = false
}

async function handleRestoreConfirm() {
  showRollbackConfirm = false
  if (!activePath) return
  rollbackBusy = true
  try {
    await api.restoreRollback(activePath)
    pushToast('已恢复回退前的状态', 'success')
    await loadProject(activePath)
    await refreshRollbackState()
  } catch (e: any) {
    const msg = String(e)
    if (msg.startsWith('CONFLICT|')) {
      await loadConflictFiles()
    } else {
      pushToast(`恢复失败: ${msg}`, 'error')
    }
  }
  rollbackBusy = false
}

async function loadConflictFiles() {
  if (!activePath) return
  try {
    conflictFiles = await api.getRollbackConflicts(activePath)
  } catch (e: any) {
    pushToast(`读取冲突失败: ${e}`, 'error')
    conflictFiles = null
  }
}

async function handleResolveConflict(choices: RollbackChoice[]) {
  if (!activePath) return
  resolvingConflict = true
  try {
    await api.resolveRollbackConflicts(activePath, choices)
    pushToast('冲突已解决，已恢复', 'success')
    conflictFiles = null
    await loadProject(activePath)
    await refreshRollbackState()
  } catch (e: any) {
    pushToast(`完成恢复失败: ${e}`, 'error')
```
